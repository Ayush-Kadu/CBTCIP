console.log("HireLeap Extension Loaded");

// Track last job to avoid duplicate popups
let lastJobText = "";

// MAIN OBSERVER
const observer = new MutationObserver(() => {
  expandJobDescription();
  detectJob();
});

observer.observe(document.body, {
  childList: true,
  subtree: true,
});

function getJobDescriptionElement() {
  return (
    // LinkedIn
    document.querySelector(".jobs-description-content__text") ||
    document.querySelector('[class*="jobs-description"]') ||
    // Indeed
    document.querySelector("#jobDescriptionText") ||
    // Naukri
    document.querySelector(".job-desc") ||
    null
  );
}

function detectJob() {
  // const jdContainer = getJobDescriptionElement();
  const jdContainer =
    document.querySelector('[class*="jobs-description"]') || // LinkedIn
    document.querySelector("#jobDescriptionText") || // Indeed
    document.querySelector('[class*="job-desc"]'); // Naukri

  if (!jdContainer) return;

  const text = jdContainer.innerText;

  // 👇 LOWER threshold (important)
  if (!text || text.length < 50) return;

  // 👇 Prevent duplicates
  if (text === lastJobText) return;

  lastJobText = text;

  // Remove old popup
  const oldPopup = document.getElementById("hireleap-popup");
  if (oldPopup) oldPopup.remove();

  // 👇 Delay ensures JD fully loads
  setTimeout(() => {
    showPopup(text);
  }, 1000);
  console.log("JD Container:", jdContainer);
}

function getUniversalJD() {
  const text = document.body.innerText;

  if (
    text.toLowerCase().includes("responsibilities") ||
    text.toLowerCase().includes("requirements") ||
    text.toLowerCase().includes("qualifications")
  ) {
    return text;
  }

  return null;
}
function getJobText() {
  const host = window.location.hostname;

  const selectors = [
    // LinkedIn
    ".jobs-description-content__text",
    ".jobs-box__html-content",
    ".jobs-description",

    // Indeed
    "#jobDescriptionText",
    ".jobsearch-JobComponent-description",
    "[data-testid='jobsearch-JobComponent-description']",

    // Naukri
    ".styles_job-desc-container__txpYf",
    ".dang-inner-html",
    ".job-desc",

    // Glassdoor
    ".jobDescriptionContent",
    ".desc",

    // Monster
    ".job-description",

    // ZipRecruiter
    ".job_description",

    // Dice
    ".job-description",

    // Internshala
    ".internship_details",

    // Generic fallback
    "main",
    "article",
  ];

  for (let selector of selectors) {
    const el = document.querySelector(selector);

    if (el && el.innerText.trim().length > 200) {
      return el.innerText.trim();
    }
  }

  return null;
}

async function handleAnalysis(jdText) {
  try {
    const authRes = await fetch("https://www.hireleap.app/api/check-auth", {
      credentials: "include",
    });

    const authData = await authRes.json();

    if (!authData.loggedIn) {
      window.open("https://www.hireleap.app/signup", "_blank");
      return;
    }

    const res = await fetch("https://www.hireleap.app/api/analyze", {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
      },
      credentials: "include",
      body: JSON.stringify({ jd: jdText }),
    });

    const result = await res.json();

    document.getElementById("result").innerHTML = `
            <p><strong>Match:</strong> ${result.score}%</p>
            <p><strong>Missing:</strong> ${result.missing.join(", ")}</p>
        `;
  } catch (err) {
    console.error("Error:", err);
  }
}
